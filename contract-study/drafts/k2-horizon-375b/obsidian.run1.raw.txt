    network: false

    read_only: false
    fs:
      may_modify:
        - "{vault}/{path}"

    network: false

        - "{vault}/{path}"
      may_modify:
        - "{vault}/**"
    expect_effect:
      fs_any_of:

    network: false

      may_create:
        - "{vault}/{destination}"
      may_delete:
        - "{vault}/{source}"
      may_modify:
        - "{vault}/**"
    expect_effect:
      fs_any_of:
        - "{vault}/{source}"
        - "{vault}/{destination}"
        - "{vault}/.obsidian-mcp/transactions/**"
    subprocess: false
    network: false
    doc_sufficient: true
    notes: "Renames {source} to {destination} and transactionally updates unambiguous backlinks in other notes; parent-directory creation for the destination is not documented; journaled per README."

    read_only: false
    fs:
      may_create:
        - "{vault}/{path}"
        - "{vault}/**"
    expect_effect:
      fs_any_of:
        - "{vault}/{path}"
        - "{vault}/.obsidian-mcp/transactions/**"
    subprocess: false
    network: false

    network: false
